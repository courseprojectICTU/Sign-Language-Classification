\chapter*{LỜI MỞ ĐẦU}
\addcontentsline{toc}{chapter}{LỜI MỞ ĐẦU}

\section*{1. Lý do chọn đề tài}
Trong những năm gần đây, sự phát triển vượt bậc của Trí tuệ nhân tạo (AI) và Học sâu (Deep Learning) đã mở ra nhiều cơ hội to lớn trong việc giải quyết các bài toán xã hội, tiêu biểu là hỗ trợ giao tiếp cho cộng đồng người khiếm thính thông qua nhận diện ngôn ngữ ký hiệu. Tuy nhiên, một thực trạng phổ biến trong môi trường học thuật là phần lớn các mô hình AI xuất sắc chỉ dừng lại ở phạm vi nghiên cứu cục bộ (chạy trên Jupyter Notebook cá nhân) mà chưa thể tiếp cận người dùng cuối do thiếu hụt hạ tầng triển khai và quy trình phát hành chuyên nghiệp.

Xuất phát từ thực tiễn trên, nhóm sinh viên chúng em đã tận dụng sản phẩm đồ án Capstone AI cuối khóa \textbf{Samsung Innovation Campus (SIC)} do thành viên tự nghiên cứu và huấn luyện để làm bài toán thực hành cốt lõi cho học phần \textbf{Điện toán đám mây}. Mục tiêu trọng tâm là vận dụng các nguyên lý tiên tiến của kỹ thuật phần mềm hiện đại: văn hóa \textbf{DevOps/MLOps}, công nghệ đóng gói \textbf{Docker Container} đa tầng, kiến trúc không máy chủ \textbf{Serverless Container} trên nền tảng \textbf{Microsoft Azure Container Apps (ACA)}, tự động hóa hạ tầng dưới dạng mã (\textbf{Terraform IaC}), kết hợp phân phối web qua \textbf{Cloudflare Pages} và tự động hóa quy trình CI/CD với \textbf{GitHub Actions} (xác thực bảo mật không mật khẩu qua \textbf{OIDC}) nhằm chuyển hóa một bài toán AI chạy cục bộ thành một dịch vụ hoạt động ổn định, tin cậy trên Internet với chi phí tối ưu.

\section*{2. Mục đích và nhiệm vụ nghiên cứu}
\begin{itemize}
    \item \textbf{Về mặt kỹ thuật đám mây và DevOps}: Thiết lập quy trình CI/CD hoàn chỉnh, tự động hóa kiểm thử mã nguồn (Unit Test, Contract Test, E2E Playwright), kiểm tra chuẩn mã nguồn (\texttt{Ruff}, \texttt{Black}), đóng gói Docker Image đa tầng và tự động phát hành an toàn lên Microsoft Azure và Cloudflare Pages khi có phiên bản mã nguồn mới.
    \item \textbf{Về mặt kiến trúc hệ thống và IaC}: Xây dựng mô hình phân tán Decoupled Architecture tối ưu tài nguyên; tự động hóa cấu hình hạ tầng đám mây bằng Terraform; giải quyết bài toán dung lượng mô hình lớn (>130 MB) và cơ chế tự động co giãn về không (Scale-to-Zero) phục vụ người dùng.
    \item \textbf{Về mặt tối ưu chi phí và vận hành}: Vận hành hệ thống ổn định trong hạn mức gói tài trợ giáo dục \textbf{Azure for Students (\$100 credit)} kết hợp hàng rào cảnh báo ngân sách tự động (\$5/tháng), đồng thời trang bị sẵn cơ chế kiểm thử khói (Smoke Test) và khôi phục tức thì (Rollback) khi có sự cố.
\end{itemize}

\section*{3. Đối tượng và phạm vi nghiên cứu}
\begin{itemize}
    \item \textbf{Đối tượng nghiên cứu}: Pipeline nhận diện 23 ký tự chữ cái VSL từ đồ án Capstone SIC; công nghệ Docker Container đa tầng; các dịch vụ Microsoft Azure (Azure Container Apps, Azure Container Registry, Log Analytics, Entra ID OIDC); Terraform; Cloudflare Pages và nền tảng tự động hóa GitHub Actions.
    \item \textbf{Phạm vi nghiên cứu}: Triển khai dịch vụ đám mây nhận ảnh tĩnh/ảnh chụp từ webcam trên trình duyệt, trả về kết quả dự đoán qua API RESTful; cô lập chặt chẽ hai môi trường Staging và Production; không mở rộng sang xử lý luồng video streaming liên tục để tối ưu tài nguyên serverless.
\end{itemize}

\section*{4. Phương pháp nghiên cứu và Ý nghĩa thực tiễn}
\begin{itemize}
    \item \textbf{Phương pháp nghiên cứu}: Phương pháp thực nghiệm kết hợp chuẩn hóa kỹ thuật phần mềm (Twelve-Factor App, Zero-Trust OIDC); đo lường độ trễ mạng, hiện tượng Cold Start, đối chiếu tương đương số học mô hình và kiểm chứng chi phí vận hành thực tế.
    \item \textbf{Ý nghĩa khoa học và thực tiễn}: Đề tài là cầu nối trực quan giữa nghiên cứu Trí tuệ nhân tạo và kỹ thuật Điện toán đám mây, cung cấp một kiến trúc tham chiếu (blueprint) chuẩn mực cho sinh viên ngành Công nghệ thông tin trong việc đưa các sản phẩm AI ra môi trường sản xuất thực tế.
\end{itemize}
